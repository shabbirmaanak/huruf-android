cmake_minimum_required(VERSION 3.22)
project(hurufwhisper C CXX)
# whisper.cpp v1.7.6 (MIT licence), copied unchanged into ./whisper.cpp; runs the Tarteel Quran model on the phone's CPU.
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
add_subdirectory(whisper.cpp)
add_library(hurufwhisper SHARED huruf_whisper.cpp)
target_compile_options(hurufwhisper PRIVATE -O3)
find_library(LOG_LIB log)
target_link_libraries(hurufwhisper whisper ${LOG_LIB})
